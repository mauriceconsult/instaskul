# M-Pesa Daraja API Configuration
# Add these to your .env file

# Get these from https://developer.safaricom.co.ke/
MPESA_CONSUMER_KEY=your_consumer_key_here
MPESA_CONSUMER_SECRET=your_consumer_secret_here

# Your business shortcode (Paybill or Till Number)
MPESA_SHORTCODE=174379  # Use your actual shortcode

# Passkey provided by Safaricom
MPESA_PASSKEY=your_passkey_here

# Callback URL - must be publicly accessible (use ngrok for local testing)
MPESA_CALLBACK_URL=https://yourdomain.com/api/mpesa/callback

# Environment: sandbox or production
MPESA_ENVIRONMENT=sandbox

# For local development with ngrok:
# 1. Install ngrok: npm install -g ngrok
# 2. Run: ngrok http 3000
# 3. Use the https URL from ngrok as your MPESA_CALLBACK_URL
# Example: MPESA_CALLBACK_URL=https://abc123.ngrok.io/api/mpesa/callback
